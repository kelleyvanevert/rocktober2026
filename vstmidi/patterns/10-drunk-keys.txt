-- Drunk keys: in beats 0, 2 and 3 every chord tone rolls its own dice on every
-- sixteenth, so the chords come out broken, strummed and never twice the same.
-- vel r4*40-60 is -60, -20, 20 or 60: half the rolls play, soft or medium.
-- Try a piano or Rhodes. Same chords as the pad.

at x.0|2|3 {
  at 5x { note e3 vel r4*40-60 dur 2  note g3 vel r4*40-60 dur 2  note b3 vel r4*40-60 dur 2  note f#4 vel r4*40-60 dur 2 }
  at 5x+1 { note c3 vel r4*40-60 dur 2  note e3 vel r4*40-60 dur 2  note b3 vel r4*40-60 dur 2  note d4 vel r4*40-60 dur 2 }
  at 5x+2 { note a2 vel r4*40-60 dur 2  note c3 vel r4*40-60 dur 2  note g3 vel r4*40-60 dur 2  note b3 vel r4*40-60 dur 2 }
  at 5x+3 { note b2 vel r4*40-60 dur 2  note d3 vel r4*40-60 dur 2  note f#3 vel r4*40-60 dur 2  note a3 vel r4*40-60 dur 2 }
  at 5x+4 { note d3 vel r4*40-60 dur 2  note a3 vel r4*40-60 dur 2  note e4 vel r4*40-60 dur 2  note f#4 vel r4*40-60 dur 2 }
}

-- the bass note of each chord, steady, on the downbeat
at 5x.0.0 { note e2 dur 6 }
at 5x+1.0.0 { note c2 dur 6 }
at 5x+2.0.0 { note a1 dur 6 }
at 5x+3.0.0 { note b1 dur 6 }
at 5x+4.0.0 { note d2 dur 6 }
